# Barrel Roll in-game verification — horizon-ctl script (one command per line).
# Drive it against a client launched with --control (or via `horizon-ctl launch`),
# inside the Xvfb recipe from the memory notes. The acceptance bar:
#   1. gliding + a mouse swipe ROLLS the camera (screenshots show the horizon tilt)
#      while the player's yaw barely moves;
#   2. A/D turns the yaw while gliding (the rudder);
#   3. landing settles the roll to level;
#   4. toggling Options > Mods > Barrel Roll off restores vanilla look.
# Assumptions to adjust per run: world name, spawn height, and the elytra slot.
launch --world BarrelRollTest --seed 1 --wait
state
give minecraft:elytra 1
give minecraft:firework_rocket 8
# climb to gliding height, then fall with the elytra equipped
tp ~ ~60 ~
# equip: put the elytra on (the chest slot swap is manual on first run — or use
# the elytra_slot mod's UI once it ships)
sleep 1
# glide: jump off and let the elytra open, then swipe the mouse sideways
key space tap
sleep 2
mouse-move 400 0
screenshot barrel-roll-1-swipe-right
state
# the roll should show; the yaw should be ~unchanged (compare with the state above)
mouse-move -800 0
screenshot barrel-roll-2-swipe-left
state
# the rudder: hold A, the yaw walks left
key a down
sleep 3
key a up
screenshot barrel-roll-3-rudder
state
# landing settles: stop gliding (touch ground) and re-check the camera
wait 6
screenshot barrel-roll-4-landed
state
